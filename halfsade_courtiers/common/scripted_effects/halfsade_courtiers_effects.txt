# Glossary for Edit
#
# Scope: recruiting ruler. Params: KEY (full position key), SKILL, SKILL2 (both get boosted),
# TA/TB/TC/TD (traits), FEMALE / MALE (force gender, yes/no)
# Aptitude scale: 3 = Good, 4 = Excellent
#
# Courtier Culture: 70% root culture, 29% other culture of root's heritage, 1% Greek (lol).
# Courtier Dynasty: 95% lowborn, 5% new generated dynasty.
# Courtier Traits: generate a character with random_traits = yes, then rebuilt:
#   - all education + personality traits are removed.
#   - education trait matching SKILL (prowess -> martial).
#   - TA, TB, TC from the decision are added.
#   - personality traits are counted; if fewer than 3, TD is tried first, then random good
#     traits are added until the candidate has exactly 3 personality traits.
#   - skill boost runs AFTER all traits, so any aptitude change is compensated.
#
# WARNING: Do not put education_* traits in TA/TB/TC.

halfsade_courtiers_recruit_effect = {
	random = {
		chance = 50
		halfsade_courtiers_try_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ TA = $TA$ TB = $TB$ TC = $TC$ TD = $TD$ FEMALE = $FEMALE$ MALE = $MALE$ LEVEL = 4 }
		halfsade_courtiers_try_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ TA = $TA$ TB = $TB$ TC = $TC$ TD = $TD$ FEMALE = $FEMALE$ MALE = $MALE$ LEVEL = 4 }
		halfsade_courtiers_try_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ TA = $TA$ TB = $TB$ TC = $TC$ TD = $TD$ FEMALE = $FEMALE$ MALE = $MALE$ LEVEL = 4 }
	}
	halfsade_courtiers_try_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ TA = $TA$ TB = $TB$ TC = $TC$ TD = $TD$ FEMALE = $FEMALE$ MALE = $MALE$ LEVEL = 3 }
	halfsade_courtiers_try_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ TA = $TA$ TB = $TB$ TC = $TC$ TD = $TD$ FEMALE = $FEMALE$ MALE = $MALE$ LEVEL = 3 }
	halfsade_courtiers_try_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ TA = $TA$ TB = $TB$ TC = $TC$ TD = $TD$ FEMALE = $FEMALE$ MALE = $MALE$ LEVEL = 3 }
	halfsade_courtiers_try_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ TA = $TA$ TB = $TB$ TC = $TC$ TD = $TD$ FEMALE = $FEMALE$ MALE = $MALE$ LEVEL = 3 }
	halfsade_courtiers_finish_effect = { KEY = $KEY$ }
}

# Attempt loop, skip if a courtier was already accepted.
halfsade_courtiers_try_effect = {
	if = {
		limit = { NOT = { exists = scope:halfsade_courtier } }

		halfsade_courtiers_pick_culture_effect = yes

		if = {
			limit = { random = 95 }

			if = {
				limit = { always = $FEMALE$ }
				create_character = {
					employer = root
					age = { 22 50 }
					gender = female
					culture = scope:halfsade_culture
					faith = root.faith
					dynasty = none
					$SKILL$ = { 14 20 }
					random_traits = yes
					save_scope_as = halfsade_candidate
				}
			}
			else_if = {
				limit = { always = $MALE$ }
				create_character = {
					employer = root
					age = { 22 50 }
					gender = male
					culture = scope:halfsade_culture
					faith = root.faith
					dynasty = none
					$SKILL$ = { 14 20 }
					random_traits = yes
					save_scope_as = halfsade_candidate
				}
			}
			else = {
				create_character = {
					employer = root
					age = { 22 50 }
					culture = scope:halfsade_culture
					faith = root.faith
					dynasty = none
					$SKILL$ = { 14 20 }
					random_traits = yes
					save_scope_as = halfsade_candidate
				}
			}
		}
		else = {
			if = {
				limit = { always = $FEMALE$ }
				create_character = {
					employer = root
					age = { 22 50 }
					gender = female
					culture = scope:halfsade_culture
					faith = root.faith
					dynasty = generate
					$SKILL$ = { 14 20 }
					random_traits = yes
					save_scope_as = halfsade_candidate
				}
			}
			else_if = {
				limit = { always = $MALE$ }
				create_character = {
					employer = root
					age = { 22 50 }
					gender = male
					culture = scope:halfsade_culture
					faith = root.faith
					dynasty = generate
					$SKILL$ = { 14 20 }
					random_traits = yes
					save_scope_as = halfsade_candidate
				}
			}
			else = {
				create_character = {
					employer = root
					age = { 22 50 }
					culture = scope:halfsade_culture
					faith = root.faith
					dynasty = generate
					$SKILL$ = { 14 20 }
					random_traits = yes
					save_scope_as = halfsade_candidate
				}
			}
		}

		if = {
			limit = { exists = scope:halfsade_candidate }

			scope:halfsade_candidate = {
				halfsade_courtiers_clear_education_effect = yes
				halfsade_courtiers_clear_personality_effect = yes

				halfsade_courtiers_edu_$SKILL$ = yes

				add_trait = $TA$
				add_trait = $TB$
				add_trait = $TC$

				if = {
					limit = { var:halfsade_pcount < 3 }
					add_trait = $TD$
				}
				halfsade_courtiers_fill_personality_effect = yes
				halfsade_courtiers_fill_personality_effect = yes
				halfsade_courtiers_fill_personality_effect = yes
				remove_variable = halfsade_pcount

				force_character_skill_recalculation = yes
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
				halfsade_courtiers_boost_effect = { KEY = $KEY$ SKILL = $SKILL$ SKILL2 = $SKILL2$ LEVEL = $LEVEL$ }
			}

			if = {
				limit = {
					scope:halfsade_candidate = {
						aptitude = {
							court_position = $KEY$
							value >= $LEVEL$
						}
						can_be_employed_as = $KEY$
					}
				}
				scope:halfsade_candidate = { save_scope_as = halfsade_courtier }
			}
			else = {
				scope:halfsade_candidate = {
					death = { death_reason = death_vanished }

# Stooge (Landless Adventurers)
# Stooge aptitude goes UP as Learning goes DOWN.
halfsade_courtiers_recruit_stooge_effect = {
	random = {
		chance = 50
		halfsade_courtiers_try_stooge_effect = { LEVEL = 4 }
		halfsade_courtiers_try_stooge_effect = { LEVEL = 4 }
		halfsade_courtiers_try_stooge_effect = { LEVEL = 4 }
	}
	halfsade_courtiers_try_stooge_effect = { LEVEL = 3 }
	halfsade_courtiers_try_stooge_effect = { LEVEL = 3 }
	halfsade_courtiers_try_stooge_effect = { LEVEL = 3 }
	halfsade_courtiers_try_stooge_effect = { LEVEL = 3 }
	halfsade_courtiers_finish_effect = { KEY = stooge_camp_officer }
}

halfsade_courtiers_try_stooge_effect = {
	if = {
		limit = { NOT = { exists = scope:halfsade_courtier } }

		halfsade_courtiers_pick_culture_effect = yes

		if = {
			limit = { random = 95 }
			create_character = {
				employer = root
				age = { 22 50 }
				culture = scope:halfsade_culture
				faith = root.faith
				dynasty = none
				learning = { 0 3 }
				random_traits = yes
				save_scope_as = halfsade_candidate
			}
		}
		else = {
			create_character = {
				employer = root
				age = { 22 50 }
				culture = scope:halfsade_culture
				faith = root.faith
				dynasty = generate
				learning = { 0 3 }
				random_traits = yes
				save_scope_as = halfsade_candidate
			}
		}

		if = {
			limit = { exists = scope:halfsade_candidate }

			scope:halfsade_candidate = {
				halfsade_courtiers_clear_education_effect = yes
				halfsade_courtiers_clear_personality_effect = yes
				remove_trait = intellect_good_1
				remove_trait = intellect_good_2
				remove_trait = intellect_good_3

				halfsade_courtiers_edu_diplomacy = yes

				add_trait = trusting
				add_trait = humble
				add_trait = content
				force_character_skill_recalculation = yes
			}

			if = {
				limit = {
					scope:halfsade_candidate = {
						aptitude = {
							court_position = stooge_camp_officer
							value >= $LEVEL$
						}
						can_be_employed_as = stooge_camp_officer
					}
				}
				scope:halfsade_candidate = { save_scope_as = halfsade_courtier }
			}
			else = {
				scope:halfsade_candidate = {
					death = { death_reason = death_vanished }
				}
			}
		}
	}
}

# Courtier Culture
halfsade_courtiers_pick_culture_effect = {
	random_list = {
		70 = {
			root.culture = { save_scope_as = halfsade_culture }
		}
		29 = {
			root.culture = { save_scope_as = halfsade_culture }
			random_culture_global = {
				limit = {
					has_same_culture_heritage = root.culture
					NOT = { this = root.culture }
					culture_number_of_counties > 0
				}
				save_scope_as = halfsade_culture
			}
		}
		1 = {
			culture:greek = { save_scope_as = halfsade_culture }
		}
	}
}

halfsade_courtiers_boost_effect = {
	if = {
		limit = {
			NOT = {
				aptitude = {
					court_position = $KEY$
					value >= $LEVEL$
				}

# Notification
halfsade_courtiers_finish_effect = {
	if = {
		limit = { exists = scope:halfsade_courtier }

		scope:halfsade_courtier = {
			add_character_flag = {
				flag = blocked_from_leaving
				years = 25
			}
		}

		if = {
			limit = { can_employ_court_position_type = $KEY$ }
			appoint_court_position = {
				recipient = scope:halfsade_courtier
				court_position = $KEY$
			}
		}
		else_if = {
			limit = { employs_court_position = $KEY$ }
			random_court_position_holder = {
				type = $KEY$
				save_scope_as = halfsade_old_holder
			}
			replace_court_position = {
				recipient = scope:halfsade_courtier
				holder = scope:halfsade_old_holder
				court_position = $KEY$
			}
			reverse_add_opinion = {
				target = scope:halfsade_old_holder
				modifier = disappointed_opinion
				opinion = -15
			}
		}

		reverse_add_opinion = {
			target = scope:halfsade_courtier
			modifier = pleased_opinion
			opinion = 20
		}

		send_interface_toast = {
			type = event_toast_effect_good
			title = halfsade_courtiers_arrived_title
			left_icon = scope:halfsade_courtier
			custom_tooltip = halfsade_courtiers_arrived_desc
			custom_tooltip = halfsade_courtiers_n_$KEY$
		}
	}
	else = {
		if = {
			limit = { has_treasury = no }
			add_gold = medium_gold_value
		}
		send_interface_toast = {
			type = event_toast_effect_bad
			title = halfsade_courtiers_failed_title
			custom_tooltip = halfsade_courtiers_failed_desc
		}
	}
}
